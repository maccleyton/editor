```python
class Stack:
    def __init__(self):
        self._items = []
    
    def is_empty(self):
        return len(self._items) == 0
    
    def push(self, item):
        self._items.append(item)
    
    def pop(self):
        if self.is_empty():
            raise IndexError("Stack is empty")
        return self._items.pop()
    
    def peek(self):
        if self.is_empty():
            raise IndexError("Stack is empty")
        return self._items[-1]
    
    def size(self):
        return len(self._items)

def is_balanced_parentheses(expression):
    stack = Stack()
    pairs = {'(': ')', '[': ']', '{': '}'}
    
    for char in expression:
        if char in pairs:  *# Opening bracket*
            stack.push(char)
        elif char in pairs.values():  *# Closing bracket*
            if stack.is_empty():
                return False
            if pairs[stack.pop()] != char:
                return False
    
    return stack.is_empty()
```

```python
def quicksort(arr, low=0, high=None):
    if high is None:
        high = len(arr) - 1
    
    if low < high:
        pivot_index = partition(arr, low, high)
        quicksort(arr, low, pivot_index - 1)
        quicksort(arr, pivot_index + 1, high)

def partition(arr, low, high):
    pivot = arr[high]
    i = low - 1
    
    for j in range(low, high):
        if arr[j] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    return i + 1

*# Versão pythônica usando list comprehensions*
def quicksort_pythonic(arr):
    if len(arr) <= 1:
        return arr
    
    pivot = arr[len(arr) // 2]
    left = [x for x in arr if x < pivot]
    middle = [x for x in arr if x == pivot]
    right = [x for x in arr if x > pivot]
    
    return quicksort_pythonic(left) + middle + quicksort_pythonic(right)
```

```python
        return arr

    left = mergesort(arr[:mid])

def merge(left, right):
    result = []
    i = j = 0
    
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            result.append(left[i])
            i += 1
        else:
            result.append(right[j])
            j += 1

    result.extend(left[i:])
    result.extend(right[j:])
```

```python
from collections import defaultdict, deque

class Graph:
    def __init__(self):
        self.graph = defaultdict(list)
    
    def add_edge(self, u, v):
        self.graph[u].append(v)
    
    def dfs(self, start):
        visited = set()
        result = []
        self._dfs_recursive(start, visited, result)
        return result
    
    def _dfs_recursive(self, vertex, visited, result):
        visited.add(vertex)
        result.append(vertex)
        
        for neighbor in self.graph[vertex]:
            if neighbor not in visited:
                self._dfs_recursive(neighbor, visited, result)
    
    def dfs_iterative(self, start):
        visited = set()
        stack = [start]
        result = []
        
        while stack:
            vertex = stack.pop()
            if vertex not in visited:
                visited.add(vertex)
                result.append(vertex)
                
                *# Add neighbors in reverse order to match recursive DFS*
                for neighbor in reversed(self.graph[vertex]):
                    if neighbor not in visited:
                        stack.append(neighbor)
        
        return result
    
    def bfs(self, start):
        visited = set()
        queue = deque([start])
        result = []
        
        visited.add(start)
        
        while queue:
            vertex = queue.popleft()
            result.append(vertex)
            
            for neighbor in self.graph[vertex]:
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append(neighbor)
        
        return result
```

**Algoritmo de Dijkstra**

```python
import heapq
from collections import defaultdict

class WeightedGraph:
    def __init__(self):
        self.graph = defaultdict(list)
    
    def add_edge(self, u, v, weight):
        self.graph[u].append((v, weight))
    
    def dijkstra(self, start):
        distances = defaultdict(lambda: float('inf'))
        distances[start] = 0
        priority_queue = [(0, start)]
        visited = set()
        
        while priority_queue:
            current_distance, current_vertex = heapq.heappop(priority_queue)
            
            if current_vertex in visited:
                continue
            
            visited.add(current_vertex)
            
            for neighbor, weight in self.graph[current_vertex]:
                new_distance = current_distance + weight
                
                if new_distance < distances[neighbor]:
                    distances[neighbor] = new_distance
                    heapq.heappush(priority_queue, (new_distance, neighbor))
        
        return dict(distances)
```

```java
            throw new IndexOutOfBoundsException("Index: " + index + ", Size: " + size);

            resize(2 * capacity);

        size++;

            throw new IndexOutOfBoundsException("Index: " + index + ", Size: " + size);

        size--;
        data[size] = null; *// Help GC*
        
        return element;
    }

        capacity = newCapacity;
        data = Arrays.copyOf(data, capacity);
```

```java
public class BinarySearchTree<T extends Comparable<T>> {
    private class TreeNode {
        T data;
        TreeNode left;
        TreeNode right;
        
        TreeNode(T data) {
            this.data = data;
            this.left = null;
            this.right = null;
        }
    }
    
    private TreeNode root;
    
    public BinarySearchTree() {
        this.root = null;
    }
    
    public void insert(T data) {
        root = insertRecursive(root, data);
    }
    
    private TreeNode insertRecursive(TreeNode node, T data) {
        if (node == null) {
            return new TreeNode(data);
        }
        
        int comparison = data.compareTo(node.data);
        if (comparison < 0) {
            node.left = insertRecursive(node.left, data);
        } else if (comparison > 0) {
            node.right = insertRecursive(node.right, data);
        }
        *// Ignore duplicates*
        
        return node;
    }
    
    public boolean search(T data) {
        return searchRecursive(root, data);
    }
    
    private boolean searchRecursive(TreeNode node, T data) {
        if (node == null) {
            return false;
        }
        
        int comparison = data.compareTo(node.data);
        if (comparison == 0) {
            return true;
        } else if (comparison < 0) {
            return searchRecursive(node.left, data);
        } else {
            return searchRecursive(node.right, data);
        }
    }
    
    public void inorderTraversal() {
        inorderRecursive(root);
    }
    
    private void inorderRecursive(TreeNode node) {
        if (node != null) {
            inorderRecursive(node.left);
            System.out.print(node.data + " ");
            inorderRecursive(node.right);
        }
    }
}
```

```java
import java.util.Comparator;

public class SortingAlgorithms {
    
    public static <T> void quickSort(T[] arr, Comparator<T> comparator) {
        quickSort(arr, 0, arr.length - 1, comparator);
    }
    
    private static <T> void quickSort(T[] arr, int low, int high, Comparator<T> comparator) {
        if (low < high) {
            int pivotIndex = partition(arr, low, high, comparator);
            quickSort(arr, low, pivotIndex - 1, comparator);
            quickSort(arr, pivotIndex + 1, high, comparator);
        }
    }
    
    private static <T> int partition(T[] arr, int low, int high, Comparator<T> comparator) {
        T pivot = arr[high];
        int i = low - 1;
        
        for (int j = low; j < high; j++) {
            if (comparator.compare(arr[j], pivot) <= 0) {
                i++;
                swap(arr, i, j);
            }
        }
        
        swap(arr, i + 1, high);
        return i + 1;
    }
    
    private static <T> void swap(T[] arr, int i, int j) {
        T temp = arr[i];
        arr[i] = arr[j];
        arr[j] = temp;
    }
    
    *// Merge Sort*
    public static <T> void mergeSort(T[] arr, Comparator<T> comparator) {
        if (arr.length <= 1) return;
        
        @SuppressWarnings("unchecked")
        T[] temp = (T[]) new Object[arr.length];
        mergeSort(arr, temp, 0, arr.length - 1, comparator);
    }
    
    private static <T> void mergeSort(T[] arr, T[] temp, int left, int right, Comparator<T> comparator) {
        if (left < right) {
            int mid = (left + right) / 2;
            mergeSort(arr, temp, left, mid, comparator);
            mergeSort(arr, temp, mid + 1, right, comparator);
            merge(arr, temp, left, mid, right, comparator);
        }
    }
    
    private static <T> void merge(T[] arr, T[] temp, int left, int mid, int right, Comparator<T> comparator) {
        *// Copy elements to temp array*
        System.arraycopy(arr, left, temp, left, right - left + 1);
        
        int i = left, j = mid + 1, k = left;
        
        while (i <= mid && j <= right) {
            if (comparator.compare(temp[i], temp[j]) <= 0) {
                arr[k++] = temp[i++];
            } else {
                arr[k++] = temp[j++];
            }
        }
        
        *// Copy remaining elements*
        while (i <= mid) {
            arr[k++] = temp[i++];
        }
        while (j <= right) {
            arr[k++] = temp[j++];
        }
    }
}
```

```java
import java.util.*;

public class WeightedGraph {
    private Map<Integer, List<Edge>> adjacencyList;
    
    private static class Edge {
        int destination;
        int weight;
        
        Edge(int destination, int weight) {
            this.destination = destination;
            this.weight = weight;
        }
    }
    
    private static class DistanceVertex implements Comparable<DistanceVertex> {
        int vertex;
        int distance;
        
        DistanceVertex(int vertex, int distance) {
            this.vertex = vertex;
            this.distance = distance;
        }
        
        @Override
        public int compareTo(DistanceVertex other) {
            return Integer.compare(this.distance, other.distance);
        }
    }
    
    public WeightedGraph() {
        this.adjacencyList = new HashMap<>();
    }
    
    public void addVertex(int vertex) {
        adjacencyList.putIfAbsent(vertex, new ArrayList<>());
    }
    
    public void addEdge(int source, int destination, int weight) {
        addVertex(source);
        addVertex(destination);
        adjacencyList.get(source).add(new Edge(destination, weight));
    }
    
    public Map<Integer, Integer> dijkstra(int start) {
        Map<Integer, Integer> distances = new HashMap<>();
        PriorityQueue<DistanceVertex> priorityQueue = new PriorityQueue<>();
        Set<Integer> visited = new HashSet<>();
        
        *// Initialize distances*
        for (int vertex : adjacencyList.keySet()) {
            distances.put(vertex, Integer.MAX_VALUE);
        }
        distances.put(start, 0);
        
        priorityQueue.offer(new DistanceVertex(start, 0));
        
        while (!priorityQueue.isEmpty()) {
            DistanceVertex current = priorityQueue.poll();
            
            if (visited.contains(current.vertex)) {
                continue;
            }
            
            visited.add(current.vertex);
            
            List<Edge> neighbors = adjacencyList.get(current.vertex);
            if (neighbors != null) {
                for (Edge edge : neighbors) {
                    int newDistance = distances.get(current.vertex) + edge.weight;
                    
                    if (newDistance < distances.get(edge.destination)) {
                        distances.put(edge.destination, newDistance);
                        priorityQueue.offer(new DistanceVertex(edge.destination, newDistance));
                    }
                }
            }
        }
        
        return distances;
    }
    
    public List<Integer> bfs(int start) {
        List<Integer> result = new ArrayList<>();
        Set<Integer> visited = new HashSet<>();
        Queue<Integer> queue = new LinkedList<>();
        
        visited.add(start);
        queue.offer(start);
        
        while (!queue.isEmpty()) {
            int vertex = queue.poll();
            result.add(vertex);
            
            List<Edge> neighbors = adjacencyList.get(vertex);
            if (neighbors != null) {
                for (Edge edge : neighbors) {
                    if (!visited.contains(edge.destination)) {
                        visited.add(edge.destination);
                        queue.offer(edge.destination);
                    }
                }
            }
        }
        
        return result;
    }
}
```

```cpp
template<typename T, typename Compare = std::less<T>>
class BinarySearchTree {
private:
    struct Node {
        T data;
        std::unique_ptr<Node> left;
        std::unique_ptr<Node> right;
        
        explicit Node(const T& value) : data(value) {}
        explicit Node(T&& value) : data(std::move(value)) {}
    };
    
    std::unique_ptr<Node> root;
    Compare comp;
    
    void insert_recursive(std::unique_ptr<Node>& node, const T& value) {
        if (!node) {
            node = std::make_unique<Node>(value);
            return;
        }
        
        if (comp(value, node->data)) {
            insert_recursive(node->left, value);
        } else if (comp(node->data, value)) {
            insert_recursive(node->right, value);
        }
        *// Ignore duplicates*
    }
    
    bool search_recursive(const std::unique_ptr<Node>& node, const T& value) const {
        if (!node) {
            return false;
        }
        
        if (comp(value, node->data)) {
            return search_recursive(node->left, value);
        } else if (comp(node->data, value)) {
            return search_recursive(node->right, value);
        } else {
            return true;
        }
    }
    
    void inorder_recursive(const std::unique_ptr<Node>& node, 
                          std::function<void(const T&)> visit) const {
        if (node) {
            inorder_recursive(node->left, visit);
            visit(node->data);
            inorder_recursive(node->right, visit);
        }
    }
    
public:
    BinarySearchTree() = default;
    
    void insert(const T& value) {
        insert_recursive(root, value);
    }
    
    void insert(T&& value) {
        insert_recursive(root, std::move(value));
    }
    
    bool search(const T& value) const {
        return search_recursive(root, value);
    }
    
    void inorder_traversal(std::function<void(const T&)> visit) const {
        inorder_recursive(root, visit);
```

**Quick Sort Template**

```cpp
#include <algorithm>
#include <iterator>
#include <functional>

template<typename RandomIt, typename Compare = std::less<>>
void quicksort(RandomIt first, RandomIt last, Compare comp = {}) {
    if (std::distance(first, last) <= 1) {
        return;
    }
    
    auto pivot_it = partition_hoare(first, last, comp);
    quicksort(first, pivot_it, comp);
    quicksort(pivot_it + 1, last, comp);
}

template<typename RandomIt, typename Compare>
RandomIt partition_hoare(RandomIt first, RandomIt last, Compare comp) {
    auto pivot = *std::prev(last);
    auto i = first;
    
    for (auto j = first; j != std::prev(last); ++j) {
        if (comp(*j, pivot) || (!comp(pivot, *j) && !comp(*j, pivot))) {
            std::iter_swap(i++, j);
        }
    }
    
    std::iter_swap(i, std::prev(last));
    return i;
}

void mergesort(RandomIt first, RandomIt last, Compare comp = {}) {
    auto distance = std::distance(first, last);
    if (distance <= 1) {
        return;
    }
    
    auto middle = std::next(first, distance / 2);
    mergesort(first, middle, comp);
    mergesort(middle, last, comp);
    std::inplace_merge(first, middle, last, comp);
}

void mergesort_with_temp(RandomIt first, RandomIt last, Compare comp = {}) {

    auto distance = std::distance(first, last);
    
    if (distance <= 1) {
        return;
    }
    
    std::vector<value_type> temp(first, last);
    mergesort_helper(temp.begin(), temp.end(), first, comp);
}

template<typename RandomIt1, typename RandomIt2, typename Compare>
void mergesort_helper(RandomIt1 temp_first, RandomIt1 temp_last, 
                     RandomIt2 dest_first, Compare comp) {

    if (distance <= 1) {
        return;
    }
    
    auto temp_middle = std::next(temp_first, distance / 2);
    auto dest_middle = std::next(dest_first, distance / 2);
    
    mergesort_helper(dest_first, dest_middle, temp_first, comp);
    mergesort_helper(dest_middle, std::next(dest_first, distance), temp_middle, comp);

}
```

```cpp
        std::priority_queue<std::pair<int, int>, 
                           std::vector<std::pair<int, int>>,
                           std::greater<>> pq;

            distances[vertex] = std::numeric_limits<int>::max();
        }
        distances[start] = 0;
        
        pq.emplace(0, start);
        
        while (!pq.empty()) {
            auto [current_dist, current_vertex] = pq.top();
            pq.pop();
            
            if (current_dist > distances[current_vertex]) {
                continue;
            }
            
            auto it = adj_list.find(current_vertex);
            if (it != adj_list.end()) {
                for (const auto& [neighbor, weight] : it->second) {
                    int new_distance = current_dist + weight;
                    
                    if (new_distance < distances[neighbor]) {
                        distances[neighbor] = new_distance;
                        pq.emplace(new_distance, neighbor);
                    }
                }
            }
        }
        
        return distances;
    }

        std::vector<int> result;
        std::unordered_set<int> visited;
        std::queue<int> queue;
        
        visited.insert(start);
        queue.push(start);
        
        while (!queue.empty()) {
            int vertex = queue.front();
            queue.pop();
            result.push_back(vertex);

                for (const auto& [neighbor, _] : it->second) {
                    if (visited.find(neighbor) == visited.end()) {
                        visited.insert(neighbor);
                        queue.push(neighbor);
                    }
                }
            }
        }
```

```jsx
            result.push(left[i++]);
        } else {
            result.push(right[j++]);
        }
    }
```

```jsx
        for (const vertex of this.adjacencyList.keys()) {
            distances.set(vertex, vertex === start ? 0 : Infinity);

            const { vertex: currentVertex, distance: currentDistance } = priorityQueue.dequeue();

            visited.add(currentVertex);
            
            const neighbors = this.adjacencyList.get(currentVertex) || [];
            for (const neighbor of neighbors) {
                const newDistance = currentDistance + neighbor.weight;
                
                if (newDistance < distances.get(neighbor.node)) {
                    distances.set(neighbor.node, newDistance);
                    priorityQueue.enqueue({ vertex: neighbor.node, distance: newDistance });
                }
            }
        }
        
        return Object.fromEntries(distances);
```

```bash
begin
    result ← 1
    base ← base mod modulus

        if exponent mod 2 = 1 then
            result ← (result × base) mod modulus
        end if
        exponent ← exponent div 2
        base ← (base × base) mod modulus
    end while

 * Algoritmo probabilístico para testes de primalidade
 */
algorithm MillerRabinTest(n: Integer, k: Integer): Boolean
precondition: n > 2, k > 0
begin
    if n = 2 or n = 3 then return true
    if n mod 2 = 0 then return false
    
    // Escrever n-1 como d × 2^r
    r ← 0
    d ← n - 1
    while d mod 2 = 0 do
        d ← d div 2
        r ← r + 1
    end while
    
    // Realizar k testes
    for i ← 1 to k do
        a ← RandomInteger(2, n-2)
        x ← ModularExponentiation(a, d, n)

            continue
        end if
        
        for j ← 1 to r-1 do
            x ← (x × x) mod n
            if x = n-1 then
                goto next_witness
            end if
        end for
        
        return false
        
        next_witness:
    end for
```

```bash
        // Encontrar pivot
        max_index ← k

                max_index ← i

        if max_index ≠ k then
            SwapRows(A, k, max_index)
            Swap(b[k], b[max_index])
        end if
        
        // Eliminação
        for i ← k+1 to n do
            if A[k,k] ≠ 0 then
                factor ← A[i,k] / A[k,k]
                for j ← k to n do
                    A[i,j] ← A[i,j] - factor × A[k,j]
                end for
                b[i] ← b[i] - factor × b[k]
            end if
        end for
    end for
    
    // Substituição reversa
    var x: Vector[n]
    for i ← n downto 1 do
        x[i] ← b[i]
        for j ← i+1 to n do
            x[i] ← x[i] - A[i,j] × x[j]

        if A[i,i] ≠ 0 then
```

```bash
/**
 * Regra do Trapézio para Integração Numérica
 * Aproxima ∫[a,b] f(x) dx
 */
algorithm TrapezoidalRule(f: Function, a: Real, b: Real, n: Integer): Real
precondition: n > 0, a < b
begin
    h ← (b - a) / n
    sum ← (f(a) + f(b)) / 2
    
    for i ← 1 to n-1 do
        x ← a + i × h
        sum ← sum + f(x)
    end for
    
    return h × sum
end

/**
 * Regra de Simpson para Integração Numérica
 * Maior precisão que regra do trapézio
 */
algorithm SimpsonsRule(f: Function, a: Real, b: Real, n: Integer): Real
precondition: n is even, n > 0, a < b
begin
    h ← (b - a) / n
    sum ← f(a) + f(b)
    
    for i ← 1 to n-1 do
        x ← a + i × h
        if i mod 2 = 1 then
            sum ← sum + 4 × f(x)
        else
            sum ← sum + 2 × f(x)
        end if
    end for
    
    return (h / 3) × sum
end

/**
 * Método de Newton-Raphson para Encontrar Raízes
 * Encontra x tal que f(x) = 0
 */
algorithm NewtonRaphson(f: Function, f_prime: Function, x0: Real, tolerance: Real, max_iterations: Integer): Real
begin
    x ← x0
    
    for i ← 1 to max_iterations do
        fx ← f(x)
        
        if |fx| < tolerance then
            return x
        end if
        
        fpx ← f_prime(x)
        if |fpx| < tolerance then
            error "Derivative too small, method may not converge"
        end if
        
        x_new ← x - fx / fpx
        
        if |x_new - x| < tolerance then
            return x_new
        end if
        
        x ← x_new
    end for
    
    error "Method did not converge within maximum iterations"
end

/**
 * Método de Bisseção para Encontrar Raízes
 * Método mais robusto, mas convergência mais lenta
 */
algorithm BisectionMethod(f: Function, a: Real, b: Real, tolerance: Real): Real
precondition: f(a) × f(b) < 0  // f tem sinais opostos nos extremos
begin
    while (b - a) / 2 > tolerance do
        c ← (a + b) / 2
        
        if f(c) = 0 then
            return c
        end if
        
        if f(a) × f(c) < 0 then
            b ← c
        else
            a ← c
        end if
    end while
    
    return (a + b) / 2
end
```

```bash
        x_new ← x - learning_rate × grad
        
        if ||x_new - x|| < tolerance then
            return x_new
        end if
        
        x ← x_new
    end for
    
    return x
end

/**
 * Método de Newton para Otimização
 * Usa informação de segunda ordem (Hessiana)
 */
algorithm NewtonOptimization(f: Function, gradient_f: Function, hessian_f: Function,
                           x0: Vector, tolerance: Real, max_iterations: Integer): Vector
begin
    x ← x0
    
    for i ← 1 to max_iterations do
        grad ← gradient_f(x)
        
        if ||grad|| < tolerance then
            return x
        end if
        
        H ← hessian_f(x)
        
        if not IsPositiveDefinite(H) then
            // Fallback to gradient descent step
            direction ← -grad
        else
            direction ← SolveLinearSystem(H, -grad)
        end if
        
        // Line search para encontrar step size ótimo
        alpha ← LineSearch(f, x, direction)
        x_new ← x + alpha × direction
        
        if ||x_new - x|| < tolerance then
            return x_new
        end if
        
        x ← x_new
    end for
    
    return x
end

/**
 * Algoritmo Simplex para Programação Linear
 * Minimiza c^T x sujeito a Ax ≤ b, x ≥ 0
 */
algorithm SimplexMethod(c: Vector, A: Matrix, b: Vector): (Vector, Real)
begin
    // Converter para forma padrão adicionando variáveis de folga
    (A_standard, c_standard) ← ConvertToStandardForm(A, c, b)
    
    // Encontrar solução básica factível inicial
    basic_variables ← FindInitialBasis(A_standard, b)
    
    while true do
        // Calcular custos reduzidos
        reduced_costs ← ComputeReducedCosts(c_standard, A_standard, basic_variables)
        
        // Teste de otimalidade
        entering_variable ← FindEnteringVariable(reduced_costs)
        if entering_variable = null then
            break  // Solução ótima encontrada
        end if
        
        // Calcular direção
        direction ← ComputeDirection(A_standard, entering_variable, basic_variables)
        
        // Teste de limitação
        leaving_variable ← FindLeavingVariable(direction, basic_variables)
        if leaving_variable = null then
            error "Problem is unbounded"
        end if
        
        // Atualizar base
        basic_variables[leaving_variable] ← entering_variable
        PerformPivot(A_standard, basic_variables, entering_variable, leaving_variable)
    end while
    
    solution ← ExtractSolution(basic_variables, A_standard, b)
    objective_value ← c_standard^T × solution
    
    return (solution, objective_value)
end
```
