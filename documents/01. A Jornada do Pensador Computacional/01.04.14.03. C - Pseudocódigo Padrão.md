## Linguagem Algorítmica Padronizada

O pseudocódigo fornece representação algorítmica independente de linguagem de programação, facilitando comunicação de ideias, análise de algoritmos e documentação técnica. Esta seção estabelece convenções padronizadas para escrita de pseudocódigo claro e preciso.

```bash
  new_node^.data ← value
  new_node^.next ← list.head
  list.head ← new_node

function Search(list: LinkedList, target: Integer): Boolean
var current: Pointer to Node
begin
```

**Pilhas (Stacks)**

```bash
type Stack = record
  data: Array[1..max_size] of Integer
  top: Integer
end record

procedure InitializeStack(var s: Stack)
begin
  s.top ← 0
end

function IsEmpty(s: Stack): Boolean
begin
  return s.top = 0
end

function IsFull(s: Stack): Boolean  
begin
  return s.top = max_size
end

procedure Push(var s: Stack, value: Integer)
begin
  if not IsFull(s) then
    s.top ← s.top + 1
    s.data[s.top] ← value
  else
    error "Stack overflow"
  end if
end

function Pop(var s: Stack): Integer
begin
  if not IsEmpty(s) then
    value ← s.data[s.top]
    s.top ← s.top - 1
    return value
  else
    error "Stack underflow"
  end if
end
```

**Filas (Queues)**

```bash
type Queue = record
  data: Array[1..max_size] of Integer
  front: Integer
  rear: Integer
  size: Integer
end record

procedure InitializeQueue(var q: Queue)
begin
  q.front ← 1
  q.rear ← 0
  q.size ← 0
end

function IsEmpty(q: Queue): Boolean
begin
  return q.size = 0
end

function IsFull(q: Queue): Boolean
begin
  return q.size = max_size
end

procedure Enqueue(var q: Queue, value: Integer)
begin
  if not IsFull(q) then
    q.rear ← (q.rear mod max_size) + 1
    q.data[q.rear] ← value
    q.size ← q.size + 1
  else
    error "Queue overflow"
  end if
end

function Dequeue(var q: Queue): Integer
begin
  if not IsEmpty(q) then
    value ← q.data[q.front]
    q.front ← (q.front mod max_size) + 1
    q.size ← q.size - 1
    return value
  else
    error "Queue underflow"
  end if
end
```

## C.4 Algoritmos Fundamentais

### Algoritmos de Ordenação

**Bubble Sort**

```bash
/**
 * Ordena array usando bubble sort
 * @complexity: O(n²) time, O(1) space
 */
procedure BubbleSort(var arr: Array[1..n] of Integer)
var i, j, temp: Integer
    swapped: Boolean
begin
  for i ← 1 to n-1 do
    swapped ← false
    for j ← 1 to n-i do
      if arr[j] > arr[j+1] then
        // Troca elementos
        temp ← arr[j]
        arr[j] ← arr[j+1]
        arr[j+1] ← temp
        swapped ← true
      end if
    end for
    // Otimização: se não houve troca, está ordenado
    if not swapped then
      break
    end if
  end for
end
```

**Quick Sort**

```bash
/**
 * Ordena array usando quicksort
 * @complexity: O(n log n) average, O(n²) worst, O(log n) space
 */
procedure QuickSort(var arr: Array[1..n] of Integer, low: Integer, high: Integer)
var pivot_index: Integer
begin
  if low < high then
    pivot_index ← Partition(arr, low, high)
    QuickSort(arr, low, pivot_index - 1)
    QuickSort(arr, pivot_index + 1, high)
  end if
end

function Partition(var arr: Array[1..n] of Integer, low: Integer, high: Integer): Integer
var pivot, i, j, temp: Integer
begin
  pivot ← arr[high]  // último elemento como pivot
  i ← low - 1
  
  for j ← low to high - 1 do
    if arr[j] <= pivot then
      i ← i + 1
      // Troca arr[i] e arr[j]
      temp ← arr[i]
      arr[i] ← arr[j]
      arr[j] ← temp
    end if
  end for
  
  // Coloca pivot na posição correta
  temp ← arr[i + 1]
  arr[i + 1] ← arr[high]
  arr[high] ← temp
  
  return i + 1
end
```

**Merge Sort**

```bash
/**
 * Ordena array usando merge sort
 * @complexity: O(n log n) time, O(n) space
 */
procedure MergeSort(var arr: Array[1..n] of Integer, left: Integer, right: Integer)
var mid: Integer
begin
  if left < right then
    mid ← (left + right) div 2
    MergeSort(arr, left, mid)
    MergeSort(arr, mid + 1, right)
    Merge(arr, left, mid, right)
  end if
end

procedure Merge(var arr: Array[1..n] of Integer, left: Integer, mid: Integer, right: Integer)
var temp: Array[left..right] of Integer
    i, j, k: Integer
begin
  // Copia elementos para array temporário
  for k ← left to right do
    temp[k] ← arr[k]
  end for
  
  i ← left
  j ← mid + 1
  k ← left
  
  // Merge dos subvetores ordenados
  while i <= mid and j <= right do
    if temp[i] <= temp[j] then
      arr[k] ← temp[i]
      i ← i + 1
    else
      arr[k] ← temp[j]
      j ← j + 1
    end if
    k ← k + 1
  end while
  
  // Copia elementos restantes
  while i <= mid do
    arr[k] ← temp[i]
    i ← i + 1
    k ← k + 1
  end while
  
  while j <= right do
    arr[k] ← temp[j]
    j ← j + 1
    k ← k + 1
  end while
end
```

```bash
/**
 * Busca elemento em array ordenado
 * @complexity: O(log n) time, O(1) space
 */
function BinarySearch(arr: Array[1..n] of Integer, target: Integer): Integer
var left, right, mid: Integer
begin
  left ← 1
  right ← n
  
  while left <= right do
    mid ← (left + right) div 2
    
    if arr[mid] = target then
      return mid
    elseif arr[mid] < target then
      left ← mid + 1
    else
      right ← mid - 1
    end if
  end while
  
  return -1  // não encontrado
end
```

```bash
type Graph = record

  visited: Array[1..V] of Boolean
end record

/**
 * DFS recursivo
 * @complexity: O(V + E) time, O(V) space
 */
procedure DFS(var graph: Graph, vertex: Integer)
begin
  graph.visited[vertex] ← true
  print vertex
  
  for each neighbor in graph.adj_list[vertex] do
    if not graph.visited[neighbor] then
      DFS(graph, neighbor)
    end if
  end for
end

// Versão iterativa usando pilha
procedure DFS_Iterative(var graph: Graph, start: Integer)
var stack: Stack of Integer
    current: Integer
begin
  InitializeStack(stack)
  Push(stack, start)
  
  while not IsEmpty(stack) do
    current ← Pop(stack)
    
    if not graph.visited[current] then
      graph.visited[current] ← true
      print current
      
      for each neighbor in graph.adj_list[current] do
        if not graph.visited[neighbor] then
          Push(stack, neighbor)
        end if
      end for
    end if
  end while
end
```

```bash
procedure BFS(var graph: Graph, start: Integer)
var queue: Queue of Integer
    current: Integer
begin

  graph.visited[start] ← true
  Enqueue(queue, start)
  
  while not IsEmpty(queue) do
    current ← Dequeue(queue)
    print current

  end while
end
```

**Algoritmo de Dijkstra**

```bash
/**
 * Menor caminho de origem única
 * @complexity: O((V + E) log V) time with binary heap
 */
procedure Dijkstra(graph: WeightedGraph, source: Integer, 
                   var distance: Array[1..V] of Integer)
var pq: PriorityQueue of (vertex, distance)
    current, neighbor: Integer
    alt_distance: Integer
begin
  // Inicializa distâncias
  for i ← 1 to V do
    distance[i] ← INFINITY
  end for
  distance[source] ← 0
  
  InitializePriorityQueue(pq)
  Insert(pq, (source, 0))
  
  while not IsEmpty(pq) do
    current ← ExtractMin(pq).vertex
    
    for each edge (current, neighbor, weight) in graph.adj_list[current] do
      alt_distance ← distance[current] + weight
      
      if alt_distance < distance[neighbor] then
        distance[neighbor] ← alt_distance
        Insert(pq, (neighbor, alt_distance))
      end if
    end for
  end while
end
```

```bash
      obs.Update(news)
    end for
  end
  
  procedure SetNews(n: String)
  begin
    news ← n
    NotifyObservers()
  end
end class
```

```bash
// Interface da estratégia
interface SortingStrategy
  procedure Sort(var data: Array of Integer)
end interface

// Estratégias concretas
class QuickSortStrategy implements SortingStrategy
  procedure Sort(var data: Array of Integer)
  begin
    QuickSort(data, 1, length(data))
  end
end class

class MergeSortStrategy implements SortingStrategy
  procedure Sort(var data: Array of Integer)
  begin
    MergeSort(data, 1, length(data))
  end
end class

// Contexto
class Sorter
private:
  strategy: SortingStrategy
  
public:
  procedure SetStrategy(s: SortingStrategy)
  begin
    strategy ← s
  end
  
  procedure ExecuteSort(var data: Array of Integer)
  begin
    if strategy ≠ null then
      strategy.Sort(data)
    else
      error "No sorting strategy set"
    end if
  end
end class

// Uso
var sorter: Sorter
var data: Array[1..100] of Integer

sorter ← new Sorter()

// Para dados pequenos
sorter.SetStrategy(new QuickSortStrategy())
sorter.ExecuteSort(data)

// Para dados que requerem estabilidade
sorter.SetStrategy(new MergeSortStrategy())
sorter.ExecuteSort(data)
```

**Tratamento de Erros**

```bash
// Usando exceções
try
  resultado ← operacao_perigosa()
catch DivisaoPorZeroException e
  print "Erro: Divisão por zero"
  resultado ← 0
catch ArgumentoInvalidoException e
  print "Erro: Argumento inválido -", e.message
  return false
finally
  cleanup_resources()
end try

// Usando códigos de erro
function SafeDivision(a: Real, b: Real, var result: Real): Boolean
begin
  if b = 0 then
    return false  // erro
  else
    result ← a / b
    return true   // sucesso
  end if
end

// Uso com verificação
var resultado: Real
if SafeDivision(10.0, 3.0, resultado) then
  print "Resultado:", resultado
else
  print "Erro na divisão"
end if
```
